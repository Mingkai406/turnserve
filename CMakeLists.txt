cmake_minimum_required(VERSION 3.24)
project(turnserve VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(TURNSERVE_LLAMA "Build the real llama.cpp backend" OFF)
option(TURNSERVE_SANITIZE "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
find_package(Threads REQUIRED)
add_library(turnserve src/runtime.cpp src/synthetic_backend.cpp)
target_include_directories(turnserve PUBLIC include)
target_link_libraries(turnserve PUBLIC Threads::Threads)
target_compile_options(turnserve PRIVATE -Wall -Wextra -Wpedantic)
if(TURNSERVE_SANITIZE)
  target_compile_options(turnserve PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(turnserve PUBLIC -fsanitize=address,undefined)
endif()
if(TURNSERVE_LLAMA)
  include(FetchContent)
  set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
  set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
  set(GGML_METAL OFF CACHE BOOL "CPU reference build" FORCE)
  set(GGML_CUDA OFF CACHE BOOL "CPU reference build" FORCE)
  set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(llama
    GIT_REPOSITORY https://github.com/ggml-org/llama.cpp.git
    GIT_TAG 19e28a27702117d8f2eb16b825b9a308111f67d9)
  FetchContent_MakeAvailable(llama)
  target_sources(turnserve PRIVATE src/llama_backend.cpp)
  target_link_libraries(turnserve PRIVATE llama)
  target_compile_definitions(turnserve PUBLIC TURNSERVE_HAS_LLAMA=1)
endif()
add_executable(turnserve-replay apps/replay.cpp)
target_link_libraries(turnserve-replay PRIVATE turnserve)
add_executable(turnserve-load apps/load.cpp)
target_link_libraries(turnserve-load PRIVATE turnserve)
if(UNIX)
  add_executable(turnserve-server apps/server.cpp)
  target_link_libraries(turnserve-server PRIVATE turnserve)
  target_compile_options(turnserve-server PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(turnserve-tests tests/runtime_test.cpp)
  target_link_libraries(turnserve-tests PRIVATE turnserve)
  if(UNIX)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_test(NAME http-contracts COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/http_contracts.py $<TARGET_FILE:turnserve-server>)
    set_tests_properties(http-contracts PROPERTIES TIMEOUT 90)
  endif()
  add_test(NAME runtime-contracts COMMAND turnserve-tests)
  add_test(NAME replay-smoke COMMAND turnserve-replay --policy interleave)
endif()
